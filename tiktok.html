function updateLikeCounter(btn, delta){ const counter = btn.parentElement.querySelector('.like-btn + .icon-text'); if(!counter) return; let value = counter.innerText.trim(); // handle formats like 1,2k let num = parseFloat(value.replace('k','').replace(',','.')); let isK = value.includes('k'); if(isNaN(num)) num = 0; let newVal = num + delta; if(newVal < 0) newVal = 0; if(newVal >= 1000){ counter.innerText = (newVal/1000).toFixed(1).replace('.',',') + 'k'; } else { counter.innerText = Math.floor(newVal); } } const videos = document.querySelectorAll('video'); videos.forEach(video => { let clickTimer = null; video.addEventListener('click', () => { const container = video.parentElement; const likeBtn = container.querySelector('.like-btn'); if(clickTimer === null){ clickTimer = setTimeout(() => { clickTimer = null; if(video.paused){ video.play(); } else { video.pause(); } }, 250); } else { clearTimeout(clickTimer); clickTimer = null; // DOUBLE TAP => LIKE if(likeBtn){ if(!likeBtn.classList.contains('like-active')){ likeBtn.classList.add('like-active'); updateLikeCounter(likeBtn, 1); } likeBtn.classList.add('like-anim'); setTimeout(() => likeBtn.classList.remove('like-anim'), 350); const rect = likeBtn.getBoundingClientRect(); const heart = document.createElement('div'); heart.classList.add('heart-float'); heart.innerText = '❤️'; heart.style.left = rect.left + rect.width / 2 + 'px'; heart.style.top = rect.top + 'px'; document.body.appendChild(heart); setTimeout(() => heart.remove(), 900); } } }); }); // Like buttons const likeButtons = document.querySelectorAll('.like-btn'); likeButtons.forEach(btn => { btn.addEventListener('click', () => { const isActive = btn.classList.contains('like-active'); btn.classList.toggle('like-active'); btn.classList.add('like-anim'); if(isActive){ updateLikeCounter(btn, -1); } else { updateLikeCounter(btn, 1); } setTimeout(() => btn.classList.remove('like-anim'), 350); const rect = btn.getBoundingClientRect(); const heart = document.createElement('div'); heart.classList.add('heart-float'); heart.innerText = '❤️'; heart.style.left = rect.left + rect.width / 2 + 'px'; heart.style.top = rect.top + 'px'; document.body.appendChild(heart); setTimeout(() => heart.remove(), 900); }); }); // Favorite buttons const favoriteButtons = document.querySelectorAll('.favorite-btn'); favoriteButtons.forEach(btn => { btn.addEventListener('click', () => { btn.classList.toggle('favorite-active'); }); }); // Comments panel const commentsPanel = document.getElementById('commentsPanel'); const commentButtons = document.querySelectorAll('.comment-btn'); const closeComments = document.getElementById('closeComments'); commentButtons.forEach(btn => { btn.addEventListener('click', () => { commentsPanel.classList.add('active'); }); }); closeComments.addEventListener('click', () => { commentsPanel.classList.remove('active'); });
